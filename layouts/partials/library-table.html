{{/* Normalizes data/library.yml (grouped by data/library_sections.yml) into the shared {sections: [{name, meta, items}]} shape and hands off to sw-index-list.html, the row-rendering partial also used by taxonomy term pages. Every row here is an external link out to the original source, hence external=true throughout. */}} {{ $sections := slice }} {{ range hugo.Data.library_sections }} {{ $sec := . }} {{ $items := where hugo.Data.library "section" $sec.name }} {{ if gt (len $items) 0 }} {{ $rows := slice }} {{ range $items }} {{ $rows = $rows | append (dict "date" .date "title" .title "kind" .type "url" .url "external" true) }} {{ end }} {{ $sections = $sections | append (dict "name" $sec.name "meta" $sec.meta "items" $rows) }} {{ end }} {{ end }} {{ partial "sw-index-list.html" (dict "sections" $sections) }}
